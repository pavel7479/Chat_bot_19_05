# the_First_Agent

Изолированный контур первого агента.

Сюда переносится только то, что относится к цепочке работы первого агента:
- запрос + история + предыдущее состояние
- нормализация ввода
- переформулирование вопроса без добавления новых фактов
- формирование prompt для модели
- разбор JSON-ответа модели
- trace/diagnostics первого агента

Папки:
- `orchestrator/` — основной orchestration-код первого агента
- `catalog/` — работа со списком intent/тем первого агента
- `preprocessing/` — нормализация и rewrite
- `parsing/` — разбор model JSON
- `context/` — prompt/context/history helpers
- `prompts/` — prompt-файлы первого агента
- `config/` — config первого агента

Не переносится внутрь этого пакета то, что является shared для всего проекта:
- общие data-модели
- общий LLM provider
- общий config loader
- общий telemetry/logger

from __future__ import annotations

import re
from pathlib import Path

import yaml

from src.core.models import TopicDefinition


class TopicCatalog:
    def __init__(self, topics_file: Path) -> None:
        self._topics = self._load(topics_file)

    @staticmethod
    def _load(topics_file: Path) -> dict[str, TopicDefinition]:
        raw = yaml.safe_load(topics_file.read_text(encoding="utf-8")) or {}
        topics: dict[str, TopicDefinition] = {}
        topics_block = raw.get("topics")
        intents_block = raw.get("intents")

        if isinstance(topics_block, list):
            for item in topics_block:
                topic = TopicDefinition(
                    id=item["id"],
                    title=item["title"],
                    description=item["description"],
                    keywords=item.get("keywords", []),
                    aliases=item.get("aliases", []),
                    anti_keywords=item.get("anti_keywords", []),
                    choose_when="",
                    not_choose_when="",
                    examples=item.get("examples", []),
                )
                topics[topic.id] = topic
            return topics

        if isinstance(intents_block, list):
            for item in intents_block:
                if not isinstance(item, dict):
                    continue
                intent_id = str(item.get("intent", "")).strip()
                if not intent_id:
                    continue
                label_ru = str(item.get("label_ru", "")).strip() or intent_id
                choose_when_points = TopicCatalog._normalize_points(item.get("choose_when", ""))
                not_choose_when_points = TopicCatalog._normalize_points(item.get("not_choose_when", ""))
                examples = TopicCatalog._normalize_examples(item.get("examples", []))
                choose_when = "\n".join(choose_when_points)
                not_choose_when = "\n".join(not_choose_when_points)
                topic = TopicDefinition(
                    id=intent_id,
                    title=label_ru,
                    description=label_ru,
                    keywords=examples or TopicCatalog._extract_quoted_phrases(choose_when),
                    aliases=[],
                    anti_keywords=TopicCatalog._extract_quoted_phrases(not_choose_when),
                    choose_when=choose_when,
                    not_choose_when=not_choose_when,
                    examples=examples,
                )
                topics[topic.id] = topic
        return topics

    @staticmethod
    def _normalize_points(value: object) -> list[str]:
        if isinstance(value, list):
            points: list[str] = []
            for item in value:
                text = str(item).strip()
                if text and text not in points:
                    points.append(text)
            return points
        text = re.sub(r"\s+", " ", str(value or "").strip())
        if not text:
            return []
        chunks = re.split(r"(?<=[.!?])\s+|;\s+", text)
        points: list[str] = []
        for chunk in chunks:
            item = chunk.strip(" -")
            if item and item not in points:
                points.append(item)
        return points or [text]

    @staticmethod
    def _normalize_examples(value: object) -> list[str]:
        if not isinstance(value, list):
            return []
        examples: list[str] = []
        for item in value:
            text = str(item).strip()
            if text and text not in examples:
                examples.append(text)
        return examples

    @staticmethod
    def _extract_quoted_phrases(text: str) -> list[str]:
        if not text:
            return []
        phrases = re.findall(r'"([^"]+)"', text)
        unique: list[str] = []
        for phrase in phrases:
            normalized = phrase.strip()
            if normalized and normalized not in unique:
                unique.append(normalized)
        return unique

    @property
    def topics(self) -> dict[str, TopicDefinition]:
        return self._topics

    def _selected_topics(self, topic_ids: list[str] | None = None) -> list[TopicDefinition]:
        if not topic_ids:
            return list(self._topics.values())
        selected: list[TopicDefinition] = []
        for topic_id in topic_ids:
            topic = self._topics.get(str(topic_id))
            if topic is not None:
                selected.append(topic)
        return selected

    def title_map(self, topic_ids: list[str] | None = None) -> dict[str, str]:
        return {topic.id: topic.title for topic in self._selected_topics(topic_ids)}

    def allowed_intents_text(self, topic_ids: list[str] | None = None) -> str:
        return "\n".join(f"- {topic.id}" for topic in self._selected_topics(topic_ids))

    def as_prompt_text(self, topic_ids: list[str] | None = None) -> str:
        blocks: list[str] = []
        selected_topics = self._selected_topics(topic_ids)
        selected_ids = {topic.id for topic in selected_topics}
        for topic in selected_topics:
            choose_when_lines = self._filter_points_for_shortlist(
                self._points_from_text(topic.choose_when or topic.description),
                selected_ids=selected_ids,
                current_topic_id=topic.id,
            )
            not_choose_when_lines = self._filter_points_for_shortlist(
                self._points_from_text(topic.not_choose_when or "-"),
                selected_ids=selected_ids,
                current_topic_id=topic.id,
            )
            examples = topic.examples[:3] or topic.keywords[:3]
            parts = [
                f"intent_id: {topic.id}",
                "choose_when:",
                *[f"- {line}" for line in choose_when_lines],
                "not_choose_when:",
                *[f"- {line}" for line in not_choose_when_lines],
            ]
            if examples:
                parts.extend(
                    [
                        "examples:",
                        *[f'- "{item}"' for item in examples],
                    ]
                )
            blocks.append("\n".join(parts))
        return "\n\n".join(blocks)

    @staticmethod
    def _points_from_text(text: str) -> list[str]:
        normalized = re.sub(r"\s+", " ", str(text or "").strip())
        if not normalized or normalized == "-":
            return ["-"]
        chunks = re.split(r"(?<=[.!?])\s+|;\s+", normalized)
        points: list[str] = []
        for chunk in chunks:
            item = chunk.strip(" -")
            if not item:
                continue
            if item not in points:
                points.append(item)
        return points or [normalized]

    def _filter_points_for_shortlist(
        self,
        points: list[str],
        *,
        selected_ids: set[str],
        current_topic_id: str,
    ) -> list[str]:
        filtered: list[str] = []
        allowed_ids = set(selected_ids)
        allowed_ids.add(current_topic_id)
        for point in points:
            foreign_reference = False
            for topic_id in self._topics:
                if topic_id in allowed_ids:
                    continue
                if f"`{topic_id}`" in point:
                    foreign_reference = True
                    break
            if not foreign_reference:
                filtered.append(point)
        return filtered or ["-"]

    def __init__(
        self,
        topics: dict[str, TopicDefinition],
        *,
        top_k: int = 8,
        out_of_scope_threshold: float = 0.32,
        brands_file_path: Path | None = None,
        facts_file_path: Path | None = None,
    ) -> None:

        self._top_k = max(int(top_k), 2)
        self._out_of_scope_threshold = float(out_of_scope_threshold)
        project_root = Path(__file__).resolve().parents[2]
        self._brands_file_path = brands_file_path or (project_root / "src/config/brands.yaml")
        self._facts_file_path = facts_file_path or (project_root / "src/config/facts.yaml")

        self._fact_aliases_by_topic = self._load_fact_aliases_by_topic(self._facts_file_path)

    def build_shortlist(
        self,
        query: str,
        history_text: str = "",
        session_state: SessionState | None = None,
        top_k: int | None = None,
    ) -> list[TopicShortlistCandidate]:
        limit = max(int(top_k or self._top_k), 2)
        raw_query = str(query or "")
        query_tokens = set(self._tokenize(raw_query))
        context_tokens = set(self._expand_context_tokens(raw_query, history_text, session_state))
        effective_tokens = self._expand_with_synonyms(query_tokens | context_tokens)
        scoring_tokens = effective_tokens - self._NON_SCORING_REPLY_TOKENS
        brand_matches = self._detect_brands(raw_query)
        if not brand_matches and self._is_short_reply(raw_query, query_tokens) and session_state and session_state.last_mentioned_brand.strip():
            brand_matches = self._detect_brands(session_state.last_mentioned_brand)
        assistant_hints = set(self._assistant_context_hints(self._extract_last_assistant_message(history_text)))
        query_signals = self._build_query_signals(
            raw_query,
            query_tokens,
            effective_tokens,
            brand_matches,
            assistant_hints,
        )

        candidates = [
            self._score_topic(
                topic=self._topics[topic_id],
                signals=self._topic_signals[topic_id],
                effective_tokens=scoring_tokens,
                query_signals=query_signals,
                brand_matches=brand_matches,
                assistant_hints=assistant_hints,
            )
            for topic_id in self._topics
        ]
        ranked = sorted(
            candidates,
            key=lambda item: (-item.score, -item.overlap, -item.jaccard, item.avoid_overlap, item.topic_id),
        )

        need_nonsense = bool(query_signals["noisy"])
        max_profile_score = max(
            (item.score for item in ranked if item.topic_id not in {"nonsense_input", "out_of_scope_request"}),
            default=0.0,
        )
        need_out_of_scope = bool(query_signals["out_of_scope_like"] or max_profile_score < self._out_of_scope_threshold)

        fallback_ids = []
        if need_nonsense:
            fallback_ids.append("nonsense_input")
        if need_out_of_scope:
            fallback_ids.append("out_of_scope_request")

        primary_limit = max(limit - len(fallback_ids), 0)
        selected = self._select_primary_candidates(ranked, primary_limit)
        selected_ids = {item.topic_id for item in selected}
        for topic_id in fallback_ids:
            candidate = next((item for item in ranked if item.topic_id == topic_id), None)
            if candidate is None or topic_id in selected_ids:
                continue
            selected.append(candidate)
            selected_ids.add(topic_id)

        return sorted(
            selected,
            key=lambda item: (-item.score, -item.overlap, -item.jaccard, item.avoid_overlap, item.topic_id),
        )[:limit]

    def build_topic_ids(
        self,
        query: str,
        history_text: str = "",
        session_state: SessionState | None = None,
        top_k: int | None = None,
    ) -> list[str]:
        return [item.topic_id for item in self.build_shortlist(query, history_text, session_state, top_k)]

    def _select_primary_candidates(
        self,
        ranked: list[TopicShortlistCandidate],
        limit: int,
    ) -> list[TopicShortlistCandidate]:
        selected: list[TopicShortlistCandidate] = []
        for item in ranked:
            if item.topic_id in {"nonsense_input", "out_of_scope_request"}:
                continue
            selected.append(item)
            if len(selected) >= limit:
                break
        return selected

    def _score_topic(
        self,
        *,
        topic: TopicDefinition,
        signals: _TopicSignals,
        effective_tokens: set[str],
        query_signals: dict[str, bool],
        brand_matches: set[str],
        assistant_hints: set[str],
    ) -> TopicShortlistCandidate:
        score = 0.0
        for field_name, weight in self._FIELD_WEIGHTS.items():
            score += weight * self._field_match_score(effective_tokens, getattr(signals, field_name))

        overlap = self._token_overlap(effective_tokens, signals.positive_tokens)
        jaccard = self._jaccard(effective_tokens, signals.positive_tokens)
        avoid_overlap = self._avoid_overlap(effective_tokens, signals.anti_tokens)

        score += 0.18 * overlap
        score += 0.10 * jaccard
        score -= 0.25 * avoid_overlap
        score += self._intent_boost(topic.id, query_signals, brand_matches, assistant_hints)

        matched_tokens = sorted((effective_tokens & signals.positive_tokens) - self._STOPWORDS)
        return TopicShortlistCandidate(
            topic_id=topic.id,
            label_ru=topic.title,
            score=round(score, 6),
            overlap=round(overlap, 6),
            jaccard=round(jaccard, 6),
            avoid_overlap=round(avoid_overlap, 6),
            matched_tokens=matched_tokens,
        )

    def _intent_boost(
        self,
        topic_id: str,
        query_signals: dict[str, bool],
        brand_matches: set[str],
        assistant_hints: set[str],
    ) -> float:
        boost = 0.0
        if query_signals["pricing_like"] and topic_id in {"epc_tariffs", "tis_tariffs"}:
            boost += 0.40
        if query_signals["refund_like"] and topic_id == "post_payment_no_access":
            boost += 0.55
        if brand_matches and topic_id == "specific_brand_check":
            boost += 0.40
        if query_signals["list_like"] and topic_id == "brand_list_request":
            boost += 0.35
        if query_signals["physical_like"] and topic_id == "physical_person_purchase":
            boost += 0.42
        if query_signals["no_private_reason_like"] and topic_id == "no_private_sales_reason":
            boost += 0.44
        if query_signals["legal_flow_like"] and topic_id == "legal_entity_purchase_flow":
            boost += 0.42
        if query_signals["payment_without_details_like"] and topic_id == "payment_without_details":
            boost += 0.48
        if query_signals["purchase_ready_like"] and topic_id == "purchase_ready":
            boost += 0.36
        if query_signals["accuracy_like"] and topic_id == "data_accuracy_doubt":
            boost += 0.50
        if query_signals["price_objection_like"] and topic_id == "price_objection":
            boost += 0.40
        if query_signals["competitor_like"] and topic_id == "competitor_choice":
            boost += 0.42
        if query_signals["api_like"] and topic_id == "api_integration":
            boost += 0.45
        if query_signals["demo_like"] and topic_id == "demo_access":
            boost += 0.45
        if query_signals["human_like"] and topic_id == "human_operator_request":
            boost += 0.45
        if query_signals["macos_like"] and topic_id == "macos_support":
            boost += 0.45
        if query_signals["noisy"] and topic_id == "nonsense_input":
            boost += 0.65
        if query_signals["out_of_scope_like"] and topic_id == "out_of_scope_request":
            boost += 0.55
        if query_signals["short_yes_like"] and "юрлиц" in assistant_hints and topic_id == "legal_entity_purchase_flow":
            boost += 0.55
        if query_signals["short_no_like"] and "юрлиц" in assistant_hints and topic_id == "physical_person_purchase":
            boost += 0.55
        if "демо" in assistant_hints and topic_id == "demo_access" and (
            query_signals["short_yes_like"] or query_signals["short_no_like"]
        ):
            boost += 0.55
        return boost

    def _build_topic_signals(self, topic: TopicDefinition) -> _TopicSignals:
        fact_aliases = self._fact_aliases_by_topic.get(topic.id, [])
        return _TopicSignals(
            id_tokens=set(self._tokenize(topic.id)),
            title_tokens=set(self._tokenize(topic.title)),
            keyword_tokens=set(self._tokenize(" ".join(topic.keywords))),
            alias_tokens=set(self._tokenize(" ".join(topic.aliases))),
            example_tokens=set(self._tokenize(" ".join(topic.examples or fact_aliases))),
            description_tokens=set(self._tokenize(topic.description)),
            choose_tokens=set(self._tokenize(topic.choose_when)),
            anti_tokens=set(self._tokenize(" ".join(topic.anti_keywords))) | set(self._tokenize(topic.not_choose_when)),
        )

    def _build_query_signals(
        self,
        raw_query: str,
        query_tokens: set[str],
        effective_tokens: set[str],
        brand_matches: set[str],
        assistant_hints: set[str],
    ) -> dict[str, bool]:
        normalized = self._normalize_text(raw_query)
        pricing_like = bool(query_tokens & self._PRICING_TOKENS) or bool(re.search(r"\b(что\s+по\s+деньгам|на\s+год|на\s+месяц)\b", normalized))
        refund_like = any(pattern.search(normalized) for pattern in self._REFUND_PATTERNS)
        list_like = any(pattern.search(normalized) for pattern in self._LIST_PATTERNS) or normalized in {"все", "всё", "а какие есть", "какие есть"}
        physical_like = "физлицо" in effective_tokens
        no_private_reason_like = physical_like and bool(re.search(r"\b(почему|зачем)\b", normalized))
        legal_flow_like = bool(effective_tokens & {"инн", "счет", "qr", "юрлиц", "компан", "реквизит"})
        payment_without_details_like = bool(effective_tokens & {"инн", "реквизит"}) and bool(re.search(r"\b(без|не|позже)\b", normalized))
        purchase_ready_like = bool(effective_tokens & {"купить", "оформ", "подключ", "оплат"})
        accuracy_like = bool(effective_tokens & {"данн", "точн", "устаревш", "актуальн", "vin"})
        price_objection_like = bool(re.search(r"\b(дорого|дороговато|скидк)\b", normalized))
        competitor_like = bool(re.search(r"\b(конкурент|дешевле\s+у\s+друг|уйду\s+к)\b", normalized))
        api_like = "api" in effective_tokens or "1с" in normalized
        demo_like = "демо" in effective_tokens or "demo" in normalized or "trial" in normalized
        human_like = bool(re.search(r"\b(менеджер|оператор|человек)\b", normalized))
        macos_like = "macos" in effective_tokens
        noisy = self._is_noise_like(raw_query, query_tokens)
        out_of_scope_like = bool(re.search(r"\b(погода|велосипед|самокат|космическ|корабл)\b", normalized))
        return {
            "pricing_like": pricing_like,
            "refund_like": refund_like,
            "list_like": list_like,
            "physical_like": physical_like,
            "no_private_reason_like": no_private_reason_like,
            "legal_flow_like": legal_flow_like,
            "payment_without_details_like": payment_without_details_like,
            "purchase_ready_like": purchase_ready_like,
            "accuracy_like": accuracy_like,
            "price_objection_like": price_objection_like,
            "competitor_like": competitor_like,
            "api_like": api_like,
            "demo_like": demo_like,
            "human_like": human_like,
            "macos_like": macos_like,
            "noisy": noisy,
            "out_of_scope_like": out_of_scope_like,
            "short_yes_like": bool(query_tokens & {"да", "yes", "являюсь"}) or normalized in {"ага", "yep", "да являюсь"},
            "short_no_like": bool(query_tokens & {"нет", "no"}) or normalized in {"неа", "nope"},
        }

    def _expand_context_tokens(
        self,
        query: str,
        history_text: str,
        session_state: SessionState | None,
    ) -> list[str]:
        query_tokens = set(self._tokenize(query))
        if not self._is_short_reply(query, query_tokens):
            return []

        hints: list[str] = []
        last_assistant = self._extract_last_assistant_message(history_text)
        hints.extend(self._assistant_context_hints(last_assistant))
        if session_state and session_state.last_mentioned_brand.strip():
            hints.extend(self._tokenize(session_state.last_mentioned_brand))
        return hints

    def _assistant_context_hints(self, last_assistant: str) -> list[str]:
        text = self._normalize_text(last_assistant)
        if not text:
            return []
        hints: list[str] = []
        if re.search(r"\b(бренд|марк|каталог)\b", text):
            hints.extend(["бренд"])
        if re.search(r"\b(tis|тис)\b", text):
            hints.extend(["tis"])
        if re.search(r"\b(epc|епс)\b", text):
            hints.extend(["epc"])
        if re.search(r"\b(демо|demo|trial)\b", text):
            hints.extend(["демо"])
        if re.search(r"\b(юр|юридическ|ип)\w*\b", text):
            hints.extend(["юрлиц"])
        if re.search(r"\b(инн|счет|сч[её]т|qr)\b", text):
            hints.extend(["инн", "счет", "qr"])
        return hints

    def _detect_brands(self, text: str) -> set[str]:
        normalized = self._normalize_text(text)
        if not normalized:
            return set()
        matches: set[str] = set()
        for alias in self._brand_aliases:
            pattern = rf"(?<![a-zа-я0-9]){re.escape(alias)}(?![a-zа-я0-9])"
            if re.search(pattern, normalized):
                matches.add(alias)
        return matches

    @classmethod
    def _is_short_reply(cls, raw_query: str, query_tokens: set[str]) -> bool:
        normalized = cls._normalize_text(raw_query)
        if normalized in cls._SHORT_REPLY_EXACT:
            return True
        return bool(query_tokens) and len(query_tokens) <= 2 and query_tokens <= cls._SHORT_REPLY_TOKENS

    @classmethod
    def _is_noise_like(cls, raw_query: str, query_tokens: set[str]) -> bool:
        normalized = cls._normalize_text(raw_query)
        if not query_tokens:
            return True
        if normalized in {".", "..", "...", "...."}:
            return True
        if len(query_tokens) == 1:
            token = next(iter(query_tokens))
            if len(token) >= 6 and len(set(token)) <= 2:
                return True
        return False

    @classmethod
    def _load_brand_aliases(cls, path: Path) -> list[str]:
        if not path.exists():
            return []
        raw = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
        aliases: list[str] = []
        for item in raw.get("brands", []):
            canonical = cls._normalize_text(str(item.get("canonical", "")))
            if canonical:
                aliases.append(canonical)
            for alias in item.get("aliases", []):
                normalized = cls._normalize_text(str(alias))
                if normalized:
                    aliases.append(normalized)
        return sorted({alias for alias in aliases if alias}, key=len, reverse=True)

    @classmethod
    def _load_fact_aliases_by_topic(cls, path: Path) -> dict[str, list[str]]:
        if not path.exists():
            return {}
        raw = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
        grouped: dict[str, list[str]] = defaultdict(list)
        for item in raw.get("facts", []):
            topic = str(item.get("topic", "")).strip()
            if not topic:
                continue
            for alias in item.get("aliases", []):
                value = str(alias).strip()
                if value:
                    grouped[topic].append(value)
        return {key: values for key, values in grouped.items()}

    @classmethod
    def _extract_last_assistant_message(cls, history_text: str) -> str:
        lines = [line.strip() for line in str(history_text).splitlines() if line.strip()]
        for line in reversed(lines):
            if line.lower().startswith("assistant:"):
                return line.split(":", 1)[1].strip()
        return ""

    @classmethod
    def _normalize_text(cls, text: str) -> str:
        return re.sub(r"\s+", " ", str(text or "").lower().replace("ё", "е")).strip()

    @classmethod
    def _tokenize(cls, text: str) -> list[str]:
        normalized = cls._normalize_text(text)
        tokens = [match.group(0) for match in cls._TOKEN_RE.finditer(normalized)]
        cleaned: list[str] = []
        for token in tokens:
            value = cls._normalize_token(token)
            if not value:
                continue
            if value in cls._STOPWORDS and value not in cls._KEEP_TOKENS:
                continue
            cleaned.append(value)
        return cleaned

    @classmethod
    def _normalize_token(cls, token: str) -> str:
        value = str(token or "").strip().lower().replace("ё", "е")
        if not value:
            return ""
        value = cls._TOKEN_NORMALIZATIONS.get(value, value)
        for suffix in ("ами", "ями", "ого", "ему", "ому", "иях", "ах", "ях", "ов", "ев", "ей", "ам", "ям", "ом", "ем"):
            if len(value) > len(suffix) + 2 and value.endswith(suffix):
                value = value[: -len(suffix)]
                break
        if len(value) > 4 and value.endswith(("а", "я", "ы", "и", "е", "у", "ю", "о")):
            value = value[:-1]
        return cls._TOKEN_NORMALIZATIONS.get(value, value)

    @classmethod
    def _expand_with_synonyms(cls, tokens: set[str]) -> set[str]:
        expanded = set(tokens)
        for token in list(tokens):
            expanded.update(cls._SAFE_SYNONYMS.get(token, set()))
        return expanded

    @staticmethod
    def _field_match_score(query_tokens: set[str], field_tokens: set[str]) -> float:
        if not query_tokens or not field_tokens:
            return 0.0
        return len(query_tokens & field_tokens) / max(min(len(query_tokens), 4), 1)

    @staticmethod
    def _token_overlap(left: set[str], right: set[str]) -> float:
        if not left or not right:
            return 0.0
        return len(left & right) / max(len(left), 1)

    @staticmethod
    def _jaccard(left: set[str], right: set[str]) -> float:
        if not left or not right:
            return 0.0
        union = left | right
        if not union:
            return 0.0
        return len(left & right) / len(union)

    @staticmethod
    def _avoid_overlap(left: set[str], anti_tokens: set[str]) -> float:
        if not left or not anti_tokens:
            return 0.0
        return len(left & anti_tokens) / max(len(left), 1)

from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml


class IntentConfigLoader:
    def __init__(self, intents_path: Path, brands_path: Path) -> None:
        self._intents_path = intents_path
        self._brands_path = brands_path

    def load(self) -> dict[str, Any]:
        intents_raw = yaml.safe_load(self._intents_path.read_text(encoding="utf-8")) or {}
        brands_raw = yaml.safe_load(self._brands_path.read_text(encoding="utf-8")) or {}
        intents_raw.setdefault("signals", {})
        intents_raw["signals"]["brand_aliases"] = self._extract_brand_aliases(brands_raw)
        return intents_raw

    @staticmethod
    def _extract_brand_aliases(brands_raw: dict[str, Any]) -> list[str]:
        aliases: list[str] = []
        for item in brands_raw.get("brands", []):
            canonical = str(item.get("canonical", "")).strip()
            if canonical:
                aliases.append(canonical.lower())
            for alias in item.get("aliases", []):
                norm = str(alias).strip().lower()
                if norm:
                    aliases.append(norm)
        deduped: list[str] = []
        for alias in aliases:
            if alias not in deduped:
                deduped.append(alias)
        return deduped

from __future__ import annotations

import re

from the_First_Agent.context.prompt_context import PromptContext


class PromptContextExtractor:
    def extract(self, prompt: str) -> PromptContext:
        history_text = self._extract_block(
            prompt,
            "Диалог:\n",
            "\n\nПерефразированная последняя фраза клиента:\n",
        )
        prompt_before_query, user_query, prompt_after_query = self._extract_user_query_parts(prompt)
        history_lines = [line.strip() for line in history_text.splitlines() if line.strip()]
        return PromptContext(
            history_text=history_text,
            user_query=user_query,
            history_lines=history_lines,
            last_assistant_message=self._last_assistant_line(history_lines),
            prompt_before_query=prompt_before_query,
            prompt_after_query=prompt_after_query,
        )

    @staticmethod
    def _extract_block(text: str, start_marker: str, end_marker: str | None) -> str:
        start_index = text.find(start_marker)
        if start_index < 0:
            return ""
        content_start = start_index + len(start_marker)
        if end_marker is None:
            return text[content_start:].strip()
        end_index = text.find(end_marker, content_start)
        if end_index < 0:
            return text[content_start:].strip()
        return text[content_start:end_index].strip()

    @staticmethod
    def _extract_user_query_parts(prompt: str) -> tuple[str, str, str]:
        end_match = re.search(r"\n\n(?:Краткий статус диалога|Контекст диалога|Формат ответа):\n", prompt)
        if not end_match:
            return "", "", ""
        query_end = end_match.start()
        marker = "Перефразированная последняя фраза клиента:\n"
        query_start = prompt.rfind(marker, 0, query_end)
        if query_start < 0:
            return "", "", ""
        content_start = query_start + len(marker)
        return prompt[:content_start], prompt[content_start:query_end].strip(), prompt[query_end:]

    @staticmethod
    def _last_assistant_line(history_lines: list[str]) -> str:
        fallback_markers = (
            "не удалось корректно сформировать ответ",
            "уточните запрос",
        )
        for line in reversed(history_lines):
            if line.lower().startswith(("assistant:", "бот:")):
                raw = line.split(":", 1)[1].strip()
                low = raw.lower()
                if any(marker in low for marker in fallback_markers):
                    continue
                return raw
        for line in reversed(history_lines):
            raw = line.split(":", 1)[1].strip() if ":" in line else line
            low = raw.lower()
            if any(marker in low for marker in fallback_markers):
                continue
            if "?" in raw:
                return raw
        if history_lines:
            raw = history_lines[-1]
            final = raw.split(":", 1)[1].strip() if ":" in raw else raw
            low = final.lower()
            if any(marker in low for marker in fallback_markers):
                return ""
            return final
        return ""

from the_First_Agent.preprocessing.input_normalizer import InputNormalizationService
from the_First_Agent.preprocessing.query_rewriter import QueryRewriteService


class TopicClassifier:
    _SCHEMA_RETRY_LIMIT = 1
    _PRICING_TOPIC_IDS = {"epc_tariffs", "tis_tariffs"}
    _SCHEMA_RETRY_PREFIX = (
        "\n\nТвоя предыдущая попытка нарушила формат JSON.\n"
        "Ошибка схемы:\n"
        "{validation_error}\n\n"
        "Исправь ответ и верни только JSON точно в этом формате:\n"
        "- intent_1 обязателен\n"
        "- intent_2 либо null, либо объект\n"
        "- внутри intent можно использовать только поля intent_id, score, reason\n"
        "- не используй поля label, intent, confidence\n"
        "- не добавляй текст вне JSON\n"
    )

    def __init__(
        self,
        llm_provider: LLMProvider,
        topic_ids: set[str],
        topic_titles_by_id: dict[str, str] | None = None,
        intents_config_path: Path | None = None,
        brands_file_path: Path | None = None,
    ) -> None:
        self._llm = llm_provider
        self._topic_ids = topic_ids
        self._topic_titles_by_id = dict(topic_titles_by_id or {})
        project_root = Path(__file__).resolve().parents[2]
        intents_path = intents_config_path or (project_root / "src/config/config_intents.yaml")
        brands_path = brands_file_path or (project_root / "src/config/brands.yaml")
        self._intent_config = IntentConfigLoader(intents_path, brands_path).load()
        self._extractor = PromptContextExtractor()
        self._brand_aliases = [str(item) for item in self._intent_config.get("signals", {}).get("brand_aliases", []) if str(item).strip()]
        self._normalizer = InputNormalizationService()
        self._rewriter = QueryRewriteService(self._brand_aliases)
        self._parser = DirectClassificationParser(self._topic_titles_by_id)

    def classify(self, prompt: str, session_state: SessionState | None = None) -> TopicClassificationResult:
        context = self._extractor.extract(prompt)
        normalized = self._normalizer.normalize(context.user_query)
        normalized_query = str(normalized["normalized_query"])
        rewrite_trace = self._rewriter.rewrite(context, session_state, normalized_query)
        rewritten_query = str(rewrite_trace["rewritten_query"])
        effective_prompt = self._rebuild_prompt(context, prompt, rewritten_query)
        parsed_result, llm_attempts = self._generate_with_schema_retry(effective_prompt)
        parsed_result = self._apply_topic_post_rules(parsed_result, normalized_query, session_state)
        topic_ids = list(parsed_result["topic_ids"]) or [self._fallback_topic_id(str(normalized["lowered_query"]))]
        parser_fallback_reason = str(parsed_result.get("fallback_reason", "")).strip()
        state_after = self._build_state_snapshot(
            session_state=session_state,
            topic_ids=topic_ids,
            context=context,
            rewritten_query=rewritten_query,
        )
        active_pipeline = [
            "extract_context",
            "normalize_input",
            "rewrite_query",
            "build_prompt",
            "llm_generate_json",
            "parse_model_json",
            "build_state_snapshot",
        ]
        state_before = session_state.as_dict() if session_state else SessionState().as_dict()
        state_after_dict = state_after.as_dict()
        confidence = float(parsed_result["confidence"])
        reason = str(parsed_result["reason"])
        intent_scores = list(parsed_result.get("intent_scores", []))
        intent_details = list(parsed_result.get("intent_details", []))
        intent_reasons = list(parsed_result.get("intent_reasons", []))
        schema_retry_used = len(llm_attempts) > 1
        if not intent_scores:
            intent_scores = [{"intent": topic_id, "score": confidence} for topic_id in topic_ids]
        first_agent_output = {
            "original_query": context.user_query,
            "normalized_query": normalized_query,
            "rewritten_query": rewritten_query,
            "prompt": effective_prompt,
            "raw_llm_response": parsed_result["raw"],
            "parsed_json": parsed_result["parsed"],
            "topic_ids": list(topic_ids),
            "intent_scores": intent_scores,
            "intent_details": intent_details,
            "intent_reasons": intent_reasons,
            "confidence": confidence,
            "reason": reason,
            "state_snapshot": state_after_dict,
            "llm_attempts": llm_attempts,
            "schema_retry_used": schema_retry_used,
        }
        first_agent_trace = {
            "pipeline_steps": [
                {
                    "step": "extract_context",
                    "actor": "PromptContextExtractor",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {},
                    "after": {"user_query": context.user_query},
                },
                {
                    "step": "normalize_input",
                    "actor": "InputNormalizationService",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {"query": context.user_query},
                    "after": {"normalized_query": normalized_query},
                },
                {
                    "step": "rewrite_query",
                    "actor": "QueryRewriteService",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {"query": normalized_query},
                    "after": {
                        "rewritten_query": rewritten_query,
                        "rewrite_mode": str(rewrite_trace.get("mode", "")),
                    },
                },
                {
                    "step": "build_prompt",
                    "actor": "TopicClassifier",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {"prompt_user_query": context.user_query},
                    "after": {"prompt_user_query": rewritten_query},
                },
                {
                    "step": "llm_generate_json",
                    "actor": "LLMProvider",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {},
                    "after": {
                        "raw_response_present": bool(str(parsed_result["raw"]).strip()),
                        "attempt_count": len(llm_attempts),
                        "schema_retry_used": schema_retry_used,
                    },
                },
                {
                    "step": "parse_model_json",
                    "actor": "DirectClassificationParser",
                    "target": "first_agent_output",
                    "status": "ok",
                    "before": {"raw_llm_response": parsed_result["raw"]},
                    "after": {
                        "topic_ids": list(topic_ids),
                        "confidence": confidence,
                        "fallback_used": bool(parsed_result["fallback_used"]),
                        "fallback_reason": parser_fallback_reason,
                        "validation_errors": list(parsed_result.get("validation_errors", [])),
                        "schema_retry_used": schema_retry_used,
                    },
                },
                {
                    "step": "build_state_snapshot",
                    "actor": "TopicClassifier",
                    "target": "state_snapshot",
                    "status": "ok",
                    "before": state_before,
                    "after": state_after_dict,
                    "diff": dict_diff(state_before, state_after_dict),
                },
            ],
        }

        return TopicClassificationResult(
            topic_ids=topic_ids,
            confidence=confidence,
            reason=reason,
            rule_trace=[],
            state_snapshot=state_after_dict,
            diagnostics={
                "active_pipeline": active_pipeline,
                "pipeline_version": "first_agent_direct_v1",

                    "post_model_topic_override": False,
                    "rewrite_enabled": True,
                    "input_normalization_enabled": True,
                },
                "raw_llm_response": parsed_result["raw"],
                "parsed_json": parsed_result["parsed"],
                "normalization_trace": normalized,
                "rewrite_trace": rewrite_trace,
                "first_agent_output": first_agent_output,
                "first_agent_data": {},
                "first_agent_trace": first_agent_trace,
                "validation_errors": list(parsed_result.get("validation_errors", [])),
                "parser_fallback_reason": parser_fallback_reason,
                "llm_attempts": llm_attempts,
                "schema_retry_used": schema_retry_used,
                "state_before": state_before,
                "state_after": state_after_dict,
                "state_diff": dict_diff(state_before, state_after_dict),
                "state_trace": [
                    {"step": "extract_context", "user_query": context.user_query},
                    {"step": "normalize_input", "normalized_query": normalized_query},
                    {
                        "step": "rewrite_query",
                        "rewrite_mode": str(rewrite_trace.get("mode", "")),
                        "rewritten_query": rewritten_query,
                        "changed": bool(rewrite_trace.get("changed", False)),
                    },
                    {"step": "build_prompt", "prompt_rewritten": bool(rewrite_trace.get("changed", False))},
                    {
                        "step": "llm_generate_json",
                        "raw_response_present": bool(str(parsed_result["raw"]).strip()),

                    },
                    {
                        "step": "parse_model_json",
                        "topics": topic_ids,

                        "fallback_reason": parser_fallback_reason,
                        "validation_errors": list(parsed_result.get("validation_errors", [])),
                    },
                    {"step": "build_state_snapshot", "last_primary_topic": topic_ids[0] if topic_ids else "out_of_scope_request"},
                ],
            },
            classifier_source="direct_json",
            fallback_reason=parser_fallback_reason if parsed_result["fallback_used"] else "",
        )

    def _apply_topic_post_rules(
        self,
        parsed_result: dict[str, object],
        normalized_query: str,
        session_state: SessionState | None,
    ) -> dict[str, object]:
        topic_ids = list(parsed_result.get("topic_ids", []))
        if not topic_ids:
            return parsed_result

        query_tokens = set(re.findall(r"[a-zA-Zа-яА-ЯёЁ0-9]+", str(normalized_query).lower().replace("ё", "е")))
        pricing_tokens = {"сколько", "цена", "стоимость", "подписка", "доступ", "деньгам"}
        refund_tokens = {"вернуть", "возврат", "деньги", "оплатил", "оплаты", "подошло", "доступа"}
        explicit_tis = {"tis", "тис"} & query_tokens
        explicit_epc = {"epc", "епс"} & query_tokens
        has_brand_context = bool(session_state and str(session_state.last_mentioned_brand).strip())

        if query_tokens & refund_tokens and set(topic_ids).issubset(self._PRICING_TOPIC_IDS):
            return self._replace_with_single_topic(
                parsed_result,
                topic_id="post_payment_no_access",
                score=0.95,
                reason="Клиент спрашивает о возврате денег или проблеме после оплаты, это post-payment кейс, а не вопрос о тарифах.",
            )

        general_pricing = bool(query_tokens & pricing_tokens) and not explicit_tis and not explicit_epc and not has_brand_context
        if general_pricing and len(topic_ids) == 1 and topic_ids[0] in self._PRICING_TOPIC_IDS:
            missing_topic = "tis_tariffs" if topic_ids[0] == "epc_tariffs" else "epc_tariffs"
            return self._append_second_topic(
                parsed_result,
                topic_id=missing_topic,
                score=0.85,
                reason="Общий вопрос о стоимости без уточнения бренда или продукта покрывает обе продуктовые категории.",
            )
        return parsed_result

    def _replace_with_single_topic(
        self,
        parsed_result: dict[str, object],
        topic_id: str,
        score: float,
        reason: str,
    ) -> dict[str, object]:
        updated = dict(parsed_result)
        updated["topic_ids"] = [topic_id]
        updated["intent_scores"] = [{"intent": topic_id, "intent_id": topic_id, "score": score}]
        updated["intent_reasons"] = [reason]
        updated["intent_details"] = [{"intent": topic_id, "intent_id": topic_id, "score": score, "reason": reason}]
        updated["confidence"] = score
        updated["reason"] = reason
        return updated

    def _append_second_topic(
        self,
        parsed_result: dict[str, object],
        topic_id: str,
        score: float,
        reason: str,
    ) -> dict[str, object]:
        updated = dict(parsed_result)
        topic_ids = list(updated.get("topic_ids", []))
        if topic_id in topic_ids or len(topic_ids) >= 2:
            return updated
        topic_ids.append(topic_id)
        updated["topic_ids"] = topic_ids

        intent_scores = list(updated.get("intent_scores", []))
        intent_scores.append({"intent": topic_id, "intent_id": topic_id, "score": score})
        updated["intent_scores"] = intent_scores[:2]

        intent_reasons = list(updated.get("intent_reasons", []))
        intent_reasons.append(reason)
        updated["intent_reasons"] = intent_reasons[:2]

        intent_details = list(updated.get("intent_details", []))
        intent_details.append({"intent": topic_id, "intent_id": topic_id, "score": score, "reason": reason})
        updated["intent_details"] = intent_details[:2]

        updated["reason"] = "; ".join(str(item).strip() for item in updated["intent_reasons"] if str(item).strip())
        return updated

    def _generate_with_schema_retry(self, effective_prompt: str) -> tuple[dict[str, object], list[dict[str, object]]]:
        attempts: list[dict[str, object]] = []
        current_prompt = effective_prompt
        for attempt_index in range(self._SCHEMA_RETRY_LIMIT + 1):
            raw_response = self._llm.generate_json(current_prompt)
            parsed_result = self._parser.parse(raw_response)
            validation_errors = list(parsed_result.get("validation_errors", []))
            fallback_reason = str(parsed_result.get("fallback_reason", ""))
            attempts.append(
                {
                    "attempt": attempt_index + 1,
                    "feedback_applied": attempt_index > 0,
                    "prompt_suffix": current_prompt[len(effective_prompt):],
                    "raw_llm_response": raw_response,
                    "parsed_json": parsed_result.get("parsed", {}),
                    "validation_errors": validation_errors,
                    "fallback_reason": fallback_reason,
                }
            )
            if fallback_reason != "schema_validation_failed":
                return parsed_result, attempts
            if attempt_index < self._SCHEMA_RETRY_LIMIT:
                current_prompt = self._build_schema_feedback_prompt(effective_prompt, validation_errors)
        return parsed_result, attempts

    @classmethod
    def _build_schema_feedback_prompt(cls, effective_prompt: str, validation_errors: list[object]) -> str:
        validation_error = str(validation_errors[0]).strip() if validation_errors else "Неизвестная ошибка схемы."
        return effective_prompt + cls._SCHEMA_RETRY_PREFIX.format(validation_error=validation_error)

    @staticmethod
    def _rebuild_prompt(context: PromptContext, original_prompt: str, rewritten_query: str) -> str:
        if not context.prompt_before_query:
            return original_prompt
        return f"{context.prompt_before_query}{rewritten_query}{context.prompt_after_query}"

    def _build_state_snapshot(
        self,
        session_state: SessionState | None,
        topic_ids: list[str],
        context: PromptContext,
        rewritten_query: str,
    ) -> SessionState:
        base = session_state.as_dict() if session_state else SessionState().as_dict()
        primary_topic = topic_ids[0] if topic_ids else "out_of_scope_request"
        secondary_topics = list(topic_ids[1:]) if len(topic_ids) > 1 else []
        base["last_primary_topic"] = primary_topic
        base["last_topic_ids"] = list(topic_ids)
        base["last_secondary_topics"] = secondary_topics
        base["last_focus_topic"] = primary_topic
        base["last_bot_question_type"] = base.get("last_question_type", "unknown")
        base["active_request_kind"] = "multi_intent" if len(topic_ids) > 1 else primary_topic
        base["last_mentioned_brand"] = self._resolve_last_brand(context, session_state, rewritten_query)
        return SessionState(**base)

    def _resolve_last_brand(
        self,
        context: PromptContext,
        session_state: SessionState | None,
        query: str,
    ) -> str:
        if session_state and session_state.last_mentioned_brand.strip():
            return session_state.last_mentioned_brand.strip()
        haystack = f"{context.history_text}\n{query}".lower()
        last_match = ""
        for alias in self._brand_aliases:
            if re.search(rf"\b{re.escape(alias)}\b", haystack):
                last_match = alias
        return last_match.title() if last_match else ""

    @staticmethod
    def _fallback_topic_id(lowered_query: str) -> str:
        if not lowered_query or re.fullmatch(r"[\W_]+", lowered_query):
            return "nonsense_input"
        return "out_of_scope_request"

from __future__ import annotations

import json

from the_First_Agent.parsing.output_schema import FirstAgentOutputSchema


class DirectClassificationParser:
    _INTENT_ID_ALIASES = {
        "human_request": "human_operator_request",
        "human_operator": "human_operator_request",
        "competitor_intent": "competitor_choice",
        "legal_entity_purchase": "legal_entity_purchase_flow",
        "specific_brand": "specific_brand_check",
        "brand_list": "brand_list_request",
        "company_info": "company_services_info",
        "mac_os": "macos_support",
    }

    def __init__(self, topic_titles_by_id: dict[str, str]) -> None:
        self._topic_titles_by_id = {
            str(topic_id).strip(): str(title).strip()
            for topic_id, title in dict(topic_titles_by_id).items()
            if str(topic_id).strip()
        }
        self._topic_ids = set(self._topic_titles_by_id.keys())
        self._normalized_topic_ids = {
            self._normalize_intent_id(topic_id): topic_id
            for topic_id in self._topic_ids
        }
        self._normalized_aliases = {
            self._normalize_intent_id(alias): canonical
            for alias, canonical in self._INTENT_ID_ALIASES.items()
            if canonical in self._topic_ids
        }

    def parse(self, raw: str) -> dict[str, object]:
        parsed, validation_error = self._safe_parse(raw)
        if validation_error:
            topics, intent_scores, intent_reasons, intent_details, extraction_meta = [], [], [], [], {
                "seen_intent_object": False,
                "seen_intent_id": False,
                "unknown_intent_ids": [],
            }
        else:
            topics, intent_scores, intent_reasons, intent_details, extraction_meta = self._extract_topics(parsed)
        reason = self._build_combined_reason(intent_reasons)
        primary_score = float(intent_scores[0]["score"]) if intent_scores else 0.0
        fallback_reason = self._resolve_fallback_reason(
            parsed=parsed,
            topics=topics,
            validation_error=validation_error,
            extraction_meta=extraction_meta,
        )
        return {
            "raw": raw,
            "parsed": parsed,
            "topic_ids": topics,
            "intent_scores": intent_scores,
            "intent_reasons": intent_reasons,
            "intent_details": intent_details,
            "confidence": primary_score,
            "reason": reason,
            "fallback_used": bool(fallback_reason),
            "fallback_reason": fallback_reason,
            "validation_errors": [validation_error] if validation_error else [],
        }

    def _extract_topics(
        self,
        parsed: dict[str, object],
    ) -> tuple[list[str], list[dict[str, object]], list[str], list[dict[str, object]], dict[str, object]]:
        topics: list[str] = []
        scores: list[dict[str, object]] = []
        reasons: list[str] = []
        details: list[dict[str, object]] = []
        seen_intent_object = False
        seen_intent_id = False
        unknown_intent_ids: list[str] = []
        for key in ("intent_1", "intent_2"):
            value = parsed.get(key)
            if not isinstance(value, dict):
                continue
            seen_intent_object = True
            raw_intent_id = str(value.get("intent_id", "")).strip()
            if raw_intent_id:
                seen_intent_id = True
            topic_id = self._canonical_intent_id(raw_intent_id)
            if not topic_id:
                if raw_intent_id:
                    unknown_intent_ids.append(raw_intent_id)
                continue
            if topic_id in topics:
                continue
            score = self._safe_score(value.get("score"))
            reason = str(value.get("reason", "")).strip() or self._build_reason_from_intent(topic_id)
            topics.append(topic_id)
            scores.append({"intent": topic_id, "score": score, "intent_id": topic_id})
            reasons.append(reason)
            details.append(
                {
                    "intent": topic_id,
                    "intent_id": topic_id,
                    "score": score,
                    "reason": reason,
                }
            )
        return (
            topics[:2],
            scores[:2],
            reasons[:2],
            details[:2],
            {
                "seen_intent_object": seen_intent_object,
                "seen_intent_id": seen_intent_id,
                "unknown_intent_ids": unknown_intent_ids[:2],
            },
        )

    @staticmethod
    def _resolve_fallback_reason(
        parsed: dict[str, object],
        topics: list[str],
        validation_error: str,
        extraction_meta: dict[str, object],
    ) -> str:
        if validation_error.startswith("SCHEMA_VALIDATION:"):
            return "schema_validation_failed"
        if validation_error or not parsed:
            return "json_parse_failed"
        if topics:
            return ""
        unknown_intent_ids = extraction_meta.get("unknown_intent_ids", [])
        if isinstance(unknown_intent_ids, list) and unknown_intent_ids:
            return "unknown_intent_id"
        if bool(extraction_meta.get("seen_intent_object")) and not bool(extraction_meta.get("seen_intent_id")):
            return "missing_intent_id"
        return "empty_topics_after_parse"

    @staticmethod
    def _build_reason_from_intent(intent_id: str) -> str:
        if intent_id:
            return f"Клиентский запрос отнесен к теме `{intent_id}`."
        return "Fallback: invalid classifier output."

    def _canonical_intent_id(self, intent_id: str) -> str:
        normalized = self._normalize_intent_id(intent_id)
        if not normalized:
            return ""
        direct = self._normalized_topic_ids.get(normalized, "")
        if direct:
            return direct
        return self._normalized_aliases.get(normalized, "")

    @staticmethod
    def _normalize_intent_id(value: str) -> str:
        return str(value or "").strip().lower().replace("-", "_").replace(" ", "_")

    @staticmethod
    def _build_combined_reason(intent_reasons: list[str]) -> str:
        normalized = [str(item).strip() for item in intent_reasons if str(item).strip()]
        if not normalized:
            return "Fallback: invalid classifier output."
        if len(normalized) == 1:
            return normalized[0]
        return "; ".join(normalized[:2])

    def _safe_parse(self, raw: str) -> tuple[dict[str, object], str]:
        cleaned = self._extract_json(raw)
        if not cleaned:
            return {}, "Пустой или нераспознанный JSON в ответе модели."
        try:
            parsed = json.loads(cleaned)
        except Exception as exc:
            return {}, str(exc)

        if not isinstance(parsed, dict):
            return {}, "SCHEMA_VALIDATION: Корневой JSON-объект должен быть объектом."

        try:
            validated = FirstAgentOutputSchema.model_validate(parsed)
            return validated.model_dump(), ""
        except Exception as exc:
            return parsed, f"SCHEMA_VALIDATION: {exc}"

    @staticmethod
    def _extract_json(raw: str) -> str:
        text = str(raw or "").strip()
        if not text:
            return ""
        decoder = json.JSONDecoder()
        for index, char in enumerate(text):
            if char != "{":
                continue
            try:
                _, end_index = decoder.raw_decode(text[index:])
                candidate = text[index:index + end_index].strip()
                if candidate.startswith("{") and candidate.endswith("}"):
                    return candidate
            except json.JSONDecodeError:
                continue
        return ""

    @staticmethod
    def _safe_score(value: object) -> float:
        try:
            parsed = float(value)
            return max(0.0, min(1.0, parsed))
        except (TypeError, ValueError):
            return 0.0

from __future__ import annotations

import re

from src.core.models import SessionState
from the_First_Agent.context.prompt_context import PromptContext


class QueryRewriteService:
    def __init__(self, brand_aliases: list[str]) -> None:
        self._brand_aliases = [alias for alias in brand_aliases if alias]

    def rewrite(
        self,
        context: PromptContext,
        session_state: SessionState | None,
        normalized_query: str,
    ) -> dict[str, object]:
        original = normalized_query
        query = normalized_query.lower()
        last_bot = context.last_assistant_message.strip()
        rewrite_mode = "direct"
        reason = "No rewrite needed."
        rewritten = original
        last_brand = self._resolve_last_brand(context, session_state)
        last_focus = session_state.last_focus_topic if session_state else "out_of_scope_request"

        if self._is_short_answer(query) and last_bot:
            rewrite_mode = "answer_to_bot_question"
            rewritten = f"Клиент отвечает '{original}' на предыдущий вопрос бота: {last_bot}"
            reason = "Short answer rewritten through the previous bot question."
        elif re.search(r"\b(на него|по нему|для него|него|неё|нее|них|нему|ней|этот|эта|эту|эти)\b", query) and last_brand:
            rewrite_mode = "anaphora_followup"
            rewritten = self._replace_anaphora_with_brand(original, last_brand)
            reason = "Anaphora rewritten using the latest brand from history."
        elif self._is_short_catalog_followup(query, last_focus, context.history_lower):
            rewrite_mode = "short_catalog_followup"
            rewritten = self._rewrite_short_catalog_followup(query, last_focus, last_brand)
            reason = "Short catalog follow-up expanded using dialogue context."
        elif self._is_checkout_pushback(query):
            rewrite_mode = "direct"
            rewritten = original
            reason = "Checkout pushback kept as-is to avoid turning resistance into purchase intent."
        elif re.search(r"\b(оформ|оплат|счет|сч[её]т|qr|купить|договор|что нужно)\w*\b", query):
            rewrite_mode = "procedural_followup"
            rewritten = f"Клиент уточняет процесс оформления: {original}"
            reason = "Procedural follow-up rewritten into a full process question."

        return {
            "mode": rewrite_mode,
            "original_query": original,
            "rewritten_query": rewritten,
            "changed": rewritten != original,
            "reason": reason,
        }

    @staticmethod
    def _is_short_answer(query: str) -> bool:
        return query.strip() in {"да", "нет", "ага", "неа", "ок", "хорошо", "являюсь"}

    @staticmethod
    def _is_checkout_pushback(query: str) -> bool:
        return any(
            marker in query
            for marker in (
                "куда ты меня оформлять собрался",
                "куда вы меня оформлять собрался",
                "куда вы меня оформлять собрались",
                "зачем вы меня оформляете",
                "зачем ты меня оформляешь",
            )
        )

    def _resolve_last_brand(self, context: PromptContext, session_state: SessionState | None) -> str:
        if session_state and session_state.last_mentioned_brand.strip():
            return session_state.last_mentioned_brand.strip()
        haystack = f"{context.history_text}\n{context.user_query}".lower()
        last_match = ""
        for alias in self._brand_aliases:
            if re.search(rf"\b{re.escape(alias)}\b", haystack):
                last_match = alias
        return last_match.title() if last_match else ""

    @staticmethod
    def _replace_anaphora_with_brand(query: str, brand: str) -> str:
        replaced = re.sub(r"\b(на него|по нему|для него)\b", f"для {brand}", query, flags=re.IGNORECASE)
        return re.sub(r"\b(него|неё|нее|них|нему|ней|этот|эта|эту|эти)\b", brand, replaced, flags=re.IGNORECASE)

    @staticmethod
    def _is_short_catalog_followup(query: str, last_focus: str, history_text: str) -> bool:
        if re.fullmatch(r"(все|всё|а?\s*какие есть|еще какие|ещ[её]|по грузовым|по легковым|по спецтехнике)", query):
            return True
        if last_focus in {"brand_list_request", "specific_brand_check", "company_services_info"}:
            return len(query.split()) <= 4 and any(token in query for token in ("какие", "все", "еще", "груз"))
        return False

    @staticmethod
    def _rewrite_short_catalog_followup(query: str, last_focus: str, last_brand: str) -> str:
        if "груз" in query:
            return "Какие каталоги есть по грузовой технике и какие бренды доступны?"
        if query in {"все", "всё"}:
            return "Какие бренды и каталоги доступны полностью?"
        if last_brand:
            return f"Какие еще каталоги доступны помимо {last_brand}?"
        if last_focus == "specific_brand_check":
            return "Какие еще бренды и каталоги есть в наличии?"
        return "Какие каталоги и бренды есть в наличии?"

class PromptBlock:
    name: str
    required_intent_ids: frozenset[str]
    text: str
    section: str


class TopicPromptBlockCatalog:
    _BACKTICK_ID_RE = re.compile(r"`([a-z_]+)`")
    _JSON_INTENT_ID_RE = re.compile(r'"intent_id"\s*:\s*"([a-z_]+)"')

    def __init__(self) -> None:
        self._known_intent_ids = self._load_known_intent_ids()
        self._blocks = self._build_blocks()

    def blocks(self) -> tuple[PromptBlock, ...]:
        return self._blocks

    def known_intent_ids(self) -> frozenset[str]:
        return frozenset(self._known_intent_ids)

    def referenced_intent_ids(self, block: PromptBlock) -> frozenset[str]:
        referenced: set[str] = set()
        for pattern in (self._BACKTICK_ID_RE, self._JSON_INTENT_ID_RE):
            for match in pattern.findall(block.text):
                if match in self._known_intent_ids:
                    referenced.add(match)
        return frozenset(referenced)

    def effective_required_intent_ids(self, block: PromptBlock) -> frozenset[str]:
        return frozenset(set(block.required_intent_ids) | set(self.referenced_intent_ids(block)))

    def block_is_compatible(self, block: PromptBlock, selected_ids: set[str]) -> bool:
        return self.effective_required_intent_ids(block).issubset(selected_ids)

    def validate_block_definitions(self) -> list[str]:
        errors: list[str] = []
        for block in self._blocks:
            declared_unknown = sorted(set(block.required_intent_ids) - self._known_intent_ids)
            if declared_unknown:
                errors.append(
                    f"{block.name}: unknown required_intent_ids {declared_unknown}"
                )
            referenced = self.referenced_intent_ids(block)

===== the_First_Agent/prompting/topic_prompt_sections_builder.py =====
from __future__ import annotations

from the_First_Agent.prompting.topic_prompt_block_catalog import PromptBlock, TopicPromptBlockCatalog


class TopicPromptSectionsBuilder:
    def __init__(self, block_catalog: TopicPromptBlockCatalog | None = None) -> None:
        self._block_catalog = block_catalog or TopicPromptBlockCatalog()

    def build_rules_text(self, allowed_intent_ids: list[str] | set[str]) -> str:
        selected_ids = set(allowed_intent_ids)
        sections: list[str] = []

        rules = self._render_section("rules", selected_ids)
        if rules:
            sections.append("Правила для текущего shortlist:\n" + "\n".join(rules))

        priorities = self._render_section("priorities", selected_ids)
        if priorities:
            sections.append(
                "Приоритеты для текущего shortlist:\n" +
                "\n".join(f"{index}. {item}" for index, item in enumerate(priorities, start=1))
            )

        conflicts = self._render_section("conflicts", selected_ids)
        if conflicts:
            sections.append("Правила разведения близких intent-ов:\n" + "\n".join(conflicts))

        return "\n\n".join(sections)

    def build_examples_text(self, allowed_intent_ids: list[str] | set[str]) -> str:
        selected_ids = set(allowed_intent_ids)
        return "\n\n".join(self._render_section("examples", selected_ids))

    def _render_section(self, section: str, selected_ids: set[str]) -> list[str]:
        rendered: list[str] = []
        for block in self._block_catalog.blocks():
            if block.section != section:
                continue
            if not self._block_catalog.block_is_compatible(block, selected_ids):
                continue
            rendered.append(block.text)
        return rendered

===== the_First_Agent/prompts/topic_classifier_prompt.txt =====
Ты классификатор намерений клиента компании, которая продает каталоги автозапчастей.

Главное правило:
- Классифицируй только последнюю фразу клиента.
- Историю используй только для восстановления смысла коротких фраз, местоимений и брендов.
- Не добавляй intent из истории, если он не нужен для ответа на последнюю фразу клиента.

Обязательные правила:
1. Используй только `intent_id` из списка ниже.
2. Не придумывай новые `intent_id`.
3. Опирайся в первую очередь на `choose_when`, `not_choose_when` и примеры ниже.
4. `nonsense_input` выбирай только для реального шума или бессмысленного ввода.
5. `out_of_scope_request` выбирай только если запрос осмысленный, но не про продукт компании.

Правило `intent_2`:
- Заполняй `intent_2` только если в последней фразе клиента есть два самостоятельных запроса, для которых нужны два разных блока ответа.
- Не добавляй `intent_2` только потому, что одна тема связана с другой.
- Специальные исключения для отдельных intent-ов смотри в правилах текущего shortlist.

{dynamic_rules_text}
Разрешённые intent_id:
{allowed_intents_text}

Компактные описания intent-ов:
{topics_text}

{dynamic_examples_text}

Диалог:
{dialog_text}

Перефразированная последняя фраза клиента:
{user_query}

Формат ответа:
{
  "intent_1": {
    "intent_id": "один intent_id из списка выше",
    "score": 0.0,
    "reason": "коротко, почему выбрана эта тема"
  },
  "intent_2": null
}

Ограничения:
- `intent_1` обязателен всегда.
- `intent_2` либо `null`, либо объект такого же формата.
- Если тем две, в `intent_1` ставь более важную тему для ответа на последнюю фразу клиента.
- Поле `intent_id` можно заполнять только значением из блока `Разрешённые intent_id`.
- Если `intent_id` не совпадает с одним из значений из блока `Разрешённые intent_id`, ответ считается неправильным.
- Нельзя сокращать, переводить или переименовывать `intent_id`; копируй его посимвольно из разрешённого списка.
- Шкала `score`:
  - `0.90-1.00` почти полная уверенность
  - `0.75-0.89` уверенный выбор
  - `0.60-0.74` умеренная уверенность
  - ниже `0.60` использовать только если тема все равно наиболее подходящая
- Не используй поля `label`, `label_ru`, `intent`, `confidence`.
- Не добавляй текст вне JSON.
